## Checker (verbatim)
```python
import hashlib, pathlib, re, sys

root = pathlib.Path.cwd()
vf = root / ".specify/specs/079-humanize-run-output/verification.md"
keys = ("sample_id", "evidence_path", "evidence_sha256", "verdict")

rows, block = [], {}
for line in vf.read_text(encoding="utf-8").splitlines():
    m = re.match(r"^(sample_id|evidence_path|evidence_sha256|verdict)=(.*)$", line)
    if not m:
        continue
    key, value = m.group(1), m.group(2).strip()
    if key == "sample_id" and block:
        rows.append(block)
        block = {}
    block[key] = value
if block:
    rows.append(block)

failures = 0
print(f"verification.md={vf.relative_to(root)}")
print(f"records={len(rows)}")
print()
for r in rows:
    sid = r.get("sample_id", "<missing>")
    missing = [k for k in keys if not r.get(k)]
    if missing:
        print(f"{sid}: FAIL missing keys: {', '.join(missing)}")
        failures += 1
        continue
    path = root / r["evidence_path"]
    exists = path.is_file()
    size = path.stat().st_size if exists else 0
    computed = hashlib.sha256(path.read_bytes()).hexdigest() if exists else "-"
    hash_match = computed == r["evidence_sha256"]
    ok = exists and size > 0 and hash_match
    print(f"sample_id={sid}")
    print(f"  evidence_path={r['evidence_path']}")
    print(f"  exists={str(exists).lower()} bytes={size} nonempty={str(size > 0).lower()}")
    print(f"  evidence_sha256(recorded)={r['evidence_sha256']}")
    print(f"  evidence_sha256(computed)={computed}")
    print(f"  hash_match={str(hash_match).lower()} verdict={r['verdict']} result={'PASS' if ok else 'FAIL'}")
    failures += 0 if ok else 1

pass_scs = sorted({r["sample_id"].split("_")[0] for r in rows if r.get("verdict") == "pass"})
expected = ["SC-002", "SC-003", "SC-005", "SC-006", "SC-007", "SC-008", "SC-009", "SC-010"]
statuses = dict(re.findall(r"(?m)^(SC-\d{3})_status=(\w+)$", vf.read_text(encoding="utf-8")))
print()
print(f"pass_records_cover={','.join(pass_scs)}")
print(f"pass_coverage_matches_expected={str(pass_scs == expected).lower()}")
print(f"SC-001_status={statuses.get('SC-001')} (no binding duty: non-pass)")
print(f"SC-004_status={statuses.get('SC-004')} (no binding duty: non-pass)")
print(f"pass_sc_without_record={','.join(sorted(s for s in expected if s not in pass_scs)) or 'none'}")
print(f"unique_sample_ids={str(len({r.get('sample_id') for r in rows}) == len(rows)).lower()}")
print(f"RESULT={'PASS' if failures == 0 and pass_scs == expected else 'FAIL'} failures={failures}")
sys.exit(0 if failures == 0 and pass_scs == expected else 1)
```
