#!/usr/bin/env python3
"""T022 scope set difference: command / dependency / process inventories, baseline vs HEAD.

Read-only. Baseline referent is a git commit; the "current" side is the working tree, read
through the same normalisation so the two sides are comparable.
"""
import re
import subprocess
import sys
from pathlib import Path

ROOT = Path("/Users/liuqiming.lqm/project/cloud-native-ai/spec-kit")
BASELINE = "ac63fe8c"

COMMAND_DIRS = (
    "templates/commands",
    ".claude/commands",
    ".github/prompts",
    ".opencode/command",
    ".qoder/commands",
)

# Process / service / scheduler / scorer carriers. A docs-and-prompts framework declares no
# running process by construction, so the inventory is the set of paths that WOULD declare one:
# unit files, and programs whose own name says scheduler / daemon / scorer. Name-based on
# purpose — it is stable across both sides and needs no content heuristics.
PROCESS_RE = re.compile(r"(scheduler|daemon|scor(e|er)|crontab)", re.I)
PROCESS_EXT = (".py", ".sh", ".js", ".ts", ".plist", ".service", ".timer", ".socket")


def git(*args: str) -> str:
    return subprocess.run(["git", *args], cwd=ROOT, capture_output=True, text=True, check=True).stdout


def baseline_paths() -> list[str]:
    out = git("ls-tree", "-r", "--name-only", BASELINE)
    return [line for line in out.splitlines() if line]


def current_paths() -> list[str]:
    out = git("ls-files")
    tracked = {line for line in out.splitlines() if line}
    walk = {
        p.relative_to(ROOT).as_posix()
        for p in ROOT.rglob("*")
        if p.is_file() and ".git" not in p.parts and "__pycache__" not in p.parts
    }
    return sorted(tracked | walk)


def command_inventory(paths: list[str]) -> tuple[set[str], dict[str, int]]:
    """Command NAME set (``speckit.`` prefix, tool suffix and extension stripped) + dir counts.

    ``templates/commands/plan.md`` and the per-tool copy ``.claude/commands/speckit.plan.md``
    are the same command; normalising both to ``plan`` keeps the union a set of *commands*, not
    a set of files.
    """
    names: set[str] = set()
    per_dir: dict[str, int] = {}
    for path in paths:
        for directory in COMMAND_DIRS:
            if not path.startswith(directory + "/") or not path.endswith(".md"):
                continue
            stem = path.rsplit("/", 1)[1][: -len(".md")]
            stem = re.sub(r"\.prompt$", "", stem)
            stem = re.sub(r"^speckit\.", "", stem)
            names.add(stem)
            per_dir[directory] = per_dir.get(directory, 0) + 1
    return names, per_dir


def dependency_inventory() -> set[str]:
    raw = (ROOT / "pyproject.toml").read_text(encoding="utf-8")
    deps: set[str] = set()
    for block in ("dependencies = [", "requires = ["):
        for match in re.finditer(re.escape(block) + r"(.*?)\]", raw, re.S):
            for line in match.group(1).splitlines():
                item = line.strip().strip(",").strip().strip('"').strip("'")
                if item and not item.startswith("#"):
                    deps.add(item)
    return deps


def dependency_inventory_at(rev: str) -> set[str]:
    raw = git("show", f"{rev}:pyproject.toml")
    deps: set[str] = set()
    for block in ("dependencies = [", "requires = ["):
        for match in re.finditer(re.escape(block) + r"(.*?)\]", raw, re.S):
            for line in match.group(1).splitlines():
                item = line.strip().strip(",").strip().strip('"').strip("'")
                if item and not item.startswith("#"):
                    deps.add(item)
    return deps


def process_inventory(paths: list[str]) -> set[str]:
    out = set()
    for path in paths:
        base = path.rsplit("/", 1)[-1]
        if PROCESS_RE.search(base) and base.endswith(PROCESS_EXT):
            out.add(path)
    return out


def show(label: str, value) -> None:
    print(f"{label}: {value}")


def main() -> int:
    base_paths = baseline_paths()
    cur_paths = current_paths()

    print(f"# baseline commit: {BASELINE}")
    print(f"# tracked files at baseline: {len(base_paths)}; current (tracked + untracked, junk excluded): {len(cur_paths)}")
    print()

    print("== A. /speckit.* command inventory ==")
    b_names, b_dirs = command_inventory(base_paths)
    c_names, c_dirs = command_inventory(cur_paths)
    show("baseline per-directory counts", dict(sorted(b_dirs.items())))
    show("current  per-directory counts", dict(sorted(c_dirs.items())))
    show("baseline command-name count", len(b_names))
    show("current  command-name count", len(c_names))
    added_cmd = sorted(c_names - b_names)
    removed_cmd = sorted(b_names - c_names)
    show("ADDED command names", added_cmd)
    show("REMOVED command names", removed_cmd)
    print()

    print("== B. dependency inventory (pyproject.toml [project].dependencies + [build-system].requires) ==")
    b_deps = dependency_inventory_at(BASELINE)
    c_deps = dependency_inventory()
    show("baseline", sorted(b_deps))
    show("current ", sorted(c_deps))
    added_dep = sorted(c_deps - b_deps)
    removed_dep = sorted(b_deps - c_deps)
    show("ADDED dependencies", added_dep)
    show("REMOVED dependencies", removed_dep)
    print()

    print("== C. process / service / scheduler / scorer inventory ==")
    b_proc = process_inventory(base_paths)
    c_proc = process_inventory(cur_paths)
    show("baseline count", len(b_proc))
    show("current  count", len(c_proc))
    show("ADDED", sorted(c_proc - b_proc))
    show("REMOVED", sorted(b_proc - c_proc))
    print()

    print("== D. the two new validators import stdlib only ==")
    import importlib.util

    stdlib = set(sys.stdlib_module_names)
    for rel in ("scripts/python/validate-decision-tensions.py",
                "scripts/python/validate-falsification-quartet.py"):
        text = (ROOT / rel).read_text(encoding="utf-8")
        modules = set()
        for line in text.splitlines():
            m = re.match(r"\s*(?:from\s+([A-Za-z_][\w.]*)|import\s+([A-Za-z_][\w.]*))", line)
            if m:
                mod = (m.group(1) or m.group(2)).split(".")[0]
                modules.add(mod)
        local = {p.stem.replace("-", "_") for p in (ROOT / "scripts/python").glob("*.py")}
        third_party = sorted(m for m in modules if m not in stdlib and m not in local
                             and m not in {"__future__"})
        show(f"{rel} imports", sorted(modules))
        show(f"{rel} non-stdlib imports", third_party)
    print()

    print("== E. verdict ==")
    total = len(added_cmd) + len(removed_cmd) + len(added_dep) + len(removed_dep) \
        + len(c_proc - b_proc) + len(b_proc - c_proc)
    show("allowed-delta set size (must be 0)", total)
    show("new mechanism count (commands + deps + processes)", total)
    verdict = "PASS: the difference lands inside the empty set" if total == 0 \
        else "FAIL: the difference is non-empty"
    print(verdict)
    return 0 if total == 0 else 1


if __name__ == "__main__":
    sys.exit(main())
