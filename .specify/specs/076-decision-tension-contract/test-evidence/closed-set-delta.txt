```python
#!/usr/bin/env python3
"""T025 closed-set delta enumeration against baseline ac63fe8c (read-only).

Same comparison unit as scenario G: ``git diff -U0`` line bodies per file, split into the
declared allowed delta set and everything else. Prints the enumeration, then the item counts.
"""
import re
import subprocess
from pathlib import Path

ROOT = Path("/Users/liuqiming.lqm/project/cloud-native-ai/spec-kit")
BASELINE = "ac63fe8c"

FILES = {
    "skills/internal/analyze-artifacts/references/detection-and-severity.md": "detection-and-severity.md",
    "shared/guidelines/critique-and-self-critique.md": "critique-and-self-critique.md",
}


def read(rel: str) -> str:
    return (ROOT / rel).read_text(encoding="utf-8")


def show(rel: str) -> str:
    return subprocess.run(["git", "show", f"{BASELINE}:{rel}"], cwd=ROOT,
                          capture_output=True, text=True, check=True).stdout


def delta(rel: str) -> tuple[list[str], list[str]]:
    out = subprocess.run(["git", "diff", "--no-color", "-U0", BASELINE, "--", rel], cwd=ROOT,
                         capture_output=True, text=True, check=True).stdout
    removed, added = [], []
    for line in out.splitlines():
        if line.startswith("---") or line.startswith("+++") or line.startswith("@@"):
            continue
        if line.startswith("-"):
            removed.append(line[1:])
        elif line.startswith("+"):
            added.append(line[1:])
    return removed, added


def section(text: str, heading: str, stop_at_h3: bool) -> list[str]:
    lines = text.splitlines()
    start = lines.index(heading)
    end = len(lines)
    for index in range(start + 1, len(lines)):
        if lines[index].startswith("## ") or (stop_at_h3 and lines[index].startswith("### ")):
            end = index
            break
    return lines[start:end]


def bullets(text: str, heading: str) -> list[str]:
    body = section(text, heading, stop_at_h3=True)
    return [line for line in body if line.startswith("- ")]


def consumer_rows(text: str, heading: str) -> list[str]:
    body = section(text, heading, stop_at_h3=True)
    rows = [line for line in body if line.strip().startswith("|")]
    return [row for row in rows if "---" not in row][1:]


total_outside = 0
for rel, label in FILES.items():
    removed, added = delta(rel)
    now = read(rel)
    base = show(rel)

    if "detection" in label:
        allowed_removed = ["## The seven passes"]
        allowed_added = ["## The eight passes"] + section(
            now, "### I. Decision Tension records", stop_at_h3=True)
    else:
        allowed_removed = []
        allowed_added = section(now, "### 产出形态:四元组(封闭四字段)", stop_at_h3=False)

    outside_removed = [line for line in removed if line not in allowed_removed]
    outside_added = [line for line in added if line not in allowed_added]
    total_outside += len(outside_removed) + len(outside_added)

    print(f"## {rel}")
    print(f"   removed lines: {len(removed)}  (inside the allowed set: "
          f"{len(removed) - len(outside_removed)}, outside: {len(outside_removed)})")
    for line in removed:
        print(f"     -{'IN ' if line in allowed_removed else 'OUT'} {line!r}")
    print(f"   added lines  : {len(added)}  (inside the allowed set: "
          f"{len(added) - len(outside_added)}, outside: {len(outside_added)})")
    for line in added:
        print(f"     +{'IN ' if line in allowed_added else 'OUT'} {line!r}")
    print()

print("## item counts (must not fall below the baseline)")
crit_base, crit_now = show("shared/guidelines/critique-and-self-critique.md"), read("shared/guidelines/critique-and-self-critique.md")
for heading in ("### 拒绝形态(封闭清单,命中即拒)", "## 边界(本纪律不做什么)"):
    b, c = bullets(crit_base, heading), bullets(crit_now, heading)
    print(f"   {heading}: baseline {len(b)} -> current {len(c)}   "
          f"{'NO REGRESSION' if len(c) >= len(b) and c[:len(b)] == b else 'REGRESSION'}")
b, c = consumer_rows(crit_base, "## 消费者清单"), consumer_rows(crit_now, "## 消费者清单")
print(f"   ## 消费者清单 data rows: baseline {len(b)} -> current {len(c)}   "
      f"{'NO REGRESSION' if len(c) >= len(b) and c[:len(b)] == b else 'REGRESSION'}")

det_base = show("skills/internal/analyze-artifacts/references/detection-and-severity.md")
det_now = read("skills/internal/analyze-artifacts/references/detection-and-severity.md")
letters_b = re.findall(r"(?m)^### ([A-Z])\. ", det_base)
letters_c = re.findall(r"(?m)^### ([A-Z])\. ", det_now)
print(f"   pass sections: baseline {letters_b} -> current {letters_c}   "
      f"{'NO REGRESSION' if set(letters_b) <= set(letters_c) else 'REGRESSION'}")
dnf_b = bullets(det_base, "## DO-NOT-FLAG list (noise control — do not report these at any severity)")
dnf_c = bullets(det_now, "## DO-NOT-FLAG list (noise control — do not report these at any severity)")
print(f"   DO-NOT-FLAG items: baseline {len(dnf_b)} -> current {len(dnf_c)}   "
      f"{'NO REGRESSION' if len(dnf_c) >= len(dnf_b) else 'REGRESSION'}")
sev_b = [v for v in ("CRITICAL", "HIGH", "MEDIUM", "LOW") if f"**{v}**" in det_base]
sev_c = [v for v in ("CRITICAL", "HIGH", "MEDIUM", "LOW") if f"**{v}**" in det_now]
print(f"   severity value domain: baseline {len(sev_b)} values -> current {len(sev_c)} values   "
      f"{'PRESERVED' if sev_b == sev_c else 'REGRESSION'}")
print()
print(f"## VERDICT")
print(f"   deleted or rewritten lines outside the allowed delta set: {total_outside}   (must be 0)")
print(f"   SC-008: {'PASS' if total_outside == 0 else 'FAIL'}")
```
