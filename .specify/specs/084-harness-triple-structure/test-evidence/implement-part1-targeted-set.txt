```bash
#!/usr/bin/env bash
# targeted-set-v1 for feature 084 / stage=implement (Part 1, T007 close-out).
# Owner of the algorithm: contracts/test-budget.md TB-05 (this is a transcription, not a new rule).
set -euo pipefail

STAGE_BASE="${1:-fa7ce0f1}"
SPEC_DIR=".specify/specs/084-harness-triple-structure"
SHORT="$(basename "$SPEC_DIR" | sed -E 's/^[0-9]+-//')"
DELIVERY=("$SHORT" "skills/external/sdd-loop-driver" "docs/reference/skills/sdd-loop-driver.md")
TEST_ROOTS=(tests/unit tests/contract tests/integration tests/scenarios)

# changed_paths = diff vs stage_base  UNION  working-tree status
changed="$(
  {
    git diff --name-only --diff-filter=ACMR "$STAGE_BASE"..HEAD
    git status --porcelain | sed -E 's/^...//; s/^"//; s/"$//'
  } | LC_ALL=C sort -u | sed '/^$/d'
)"

candidates="$(
  for root in "${TEST_ROOTS[@]}"; do
    [ -d "$root" ] && find "$root" -type f -name 'test_*.py'
  done | LC_ALL=C sort -u
)"

selected="$(
  {
    # (a) every changed path outside test_roots: fixed-string search for the path and its module stem
    while IFS= read -r path; do
      case "$path" in tests/*) continue ;; esac
      stem="$(basename "$path")"; stem="${stem%.*}"
      printf '%s\n' "$candidates" | xargs grep -lF -- "$path" 2>/dev/null || true
      printf '%s\n' "$candidates" | xargs grep -lF -- "$stem" 2>/dev/null || true
    done <<<"$changed"
    # (b) delivery-surface contract tests: feature short name or delivery path
    for needle in "${DELIVERY[@]}"; do
      printf '%s\n' "$candidates" | xargs grep -lF -- "$needle" 2>/dev/null || true
    done
    # (c) a changed path already under test_roots selects itself
    while IFS= read -r path; do
      case "$path" in tests/unit/*|tests/contract/*|tests/integration/*|tests/scenarios/*)
        printf '%s\n' "$path" ;;
      esac
    done <<<"$changed"
  } | LC_ALL=C sort -u | sed '/^$/d'
)"

echo "# stage_base=$STAGE_BASE spec_dir=$SPEC_DIR short_name=$SHORT"
echo "# changed_paths ($(printf '%s\n' "$changed" | wc -l | tr -d ' ')):"
printf '%s\n' "$changed" | sed 's/^/#   /'
echo "# selected_paths ($(printf '%s\n' "$selected" | wc -l | tr -d ' ')):"
printf '%s\n' "$selected"
```
