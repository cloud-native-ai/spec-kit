{{- /* Nav / link label with a first-

fallback. Overrides the theme's partial. Documents in this space carry no YAML front matter, so .Title and .LinkTitle are empty; the theme would then fall back to a humanized file name, which loses the real title ("Documentation Model (What & Why)") and mangles non-English ones. The H1 is read from .RawContent rather than .Content because the theme's heading render hook appends an anchor link, so plainifying a rendered

yields a trailing "#". Machine-generated section pages are labelled by the managed `title` cascade in hugo.toml, which lands in .Title above. */ -}} {{- $title := "" -}} {{- if .LinkTitle -}} {{- $title = .LinkTitle -}} {{- else if .Title -}} {{- $title = .Title -}} {{- else -}} {{- with findRE `(?m)^#[ \t]+(.+?)[ \t]*$` .RawContent 1 -}} {{- $title = index . 0 | strings.TrimPrefix "#" | strings.TrimSpace | replaceRE "[`*_]" "" -}} {{- end -}} {{- end -}} {{- if not $title -}} {{- with .File -}} {{- $base := cond $.IsSection (path.Base .Dir) .BaseFileName -}} {{- $title = replaceRE `[-_]+` " " $base | strings.FirstUpper -}} {{- end -}} {{- end -}} {{- return $title -}}
